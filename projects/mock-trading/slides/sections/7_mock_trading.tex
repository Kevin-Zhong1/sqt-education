% Outer feed container — single tcolorbox, items sit inside with no inter-item gaps
\newenvironment{feedbox}{%
  \begin{tcolorbox}[
    enhanced, sharp corners,
    colback=gray!8, colframe=gray!25,
    borderline west={3pt}{0pt}{sqt-themecolour},
    boxrule=0.4pt, arc=0pt,
    left=5pt, right=5pt, top=0pt, bottom=0pt,
    before skip=2pt, after skip=2pt,
  ]%
  \setlength{\parskip}{0pt}\setlength{\parindent}{0pt}%
}{%
  \end{tcolorbox}%
}

% Thin separator between items
\newcommand{\feedsep}{%
  \noindent\textcolor{gray!35}{\rule{\linewidth}{0.4pt}}\par%
}

% Base row layout: {ts-color}{ts}{text-block}{source-color}{source}
\newcommand{\feedrowbase}[5]{%
  \vspace{4pt}\noindent%
  \makebox[1.3cm][l]{\textcolor{#1}{\tiny\ttfamily #2}}%
  \begin{minipage}[t]{\dimexpr\linewidth-4.5cm\relax}#3\end{minipage}%
  \hfill%
  \begin{minipage}[t]{3cm}\raggedleft\textcolor{#4}{\tiny\itshape #5}\end{minipage}%
  \par\vspace{4pt}%
}

% \feedrow{ts}{text}{source}
\newcommand{\feedrow}[3]{%
  \feedrowbase{gray!60}{#1}{\footnotesize\raggedright #2}{gray!70}{#3}%
}

% \feedrowdim{ts}{text}{source}
\newcommand{\feedrowdim}[3]{%
  \feedrowbase{gray!30}{#1}{\footnotesize\color{gray!40}\raggedright #2}{gray!30}{#3}%
}

% \feedrowann{ts}{text}{source}{anncolor}{annotation}
\newcommand{\feedrowann}[5]{%
  \feedrowbase{gray!60}{#1}{%
    \footnotesize\raggedright #2\\[2pt]%
    \textcolor{#4}{\scriptsize\bfseries\itshape $\to$ #5}%
  }{gray!70}{#3}%
}

% Feed header bar
\newcommand{\feedheader}[1]{%
  \noindent\colorbox{sqt-themecolour!85}{\makebox[\linewidth][l]{%
    \textcolor{white}{\scriptsize\bfseries\hspace{4pt}\textbullet\enspace LIVE MARKET FEED\enspace%
    \normalfont\itshape -- #1}%
  }}%
}
